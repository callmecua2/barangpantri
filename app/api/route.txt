import { NextResponse } from 'next/server';
import prisma from '@/lib/prisma';

export async function GET() {
  try {
    const barangs = await prisma.barang.findMany({
      orderBy: { nama: 'asc' },
      include: {
        movements: {
          orderBy: { tanggal: 'desc' },
          take: 1,
          include: {
            penanggungJawab: { select: { nama: true } },
          },
        },
      },
    });

    const data = barangs.map((b) => {
      const last = b.movements[0];
      return {
        id: b.id,
        nama: b.nama,
        stok: b.stok,
        updateTerakhir: last
          ? {
              tipe: last.tipe,
              jumlah: last.jumlah,
              stokSebelum: last.stokSebelum,
              stokSesudah: last.stokSesudah,
              tanggal: last.tanggal,
              penanggungJawab: last.penanggungJawab.nama,
            }
          : null,
      };
    });

    return NextResponse.json({ message: 'Berhasil', data }, { status: 200 });
  } catch (error) {
    console.error('Error /api/stock:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
